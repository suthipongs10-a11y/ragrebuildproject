#event "MilestoneEvent" //this disables this file if MilestoneEvent is not set in OperationConfig section of appsettings

macro TeleportCase(%map, %x, %y, %zeny)
{
	if(GetZeny() < %zeny)
	{
		//Dialog(name, "Sorry, you currently don't have enough zeny to use this teleport.");
		//return;
	}
	//DropZeny(%zeny);
	MoveTo(%map, %x, %y);
	return;
}

Npc("prt_fild08", "Kafra Staff", "4_F_KAFRA1", 158, 362, S) 
{
	RegisterAsKafra();
	RegisterGlobalSignal("MilestoneKafra");
	//present from the first boot rather than unlocked at level 10 - see the note on
	//EventRevealedNpc in MilestoneEvent.txt
	localstr name = "Kafra Defolty - พนักงาน Kafra";
OnSignal:
	if(Signal == "RevealOnServerUp") ShowNpc();	else StartTimer(1s);
OnTimer 20s:
	ShowNpc();
	PlayEffectAtMyLocation("PuffOfSmoke");
OnTimer 22s:
	ServerWideMessage("<color=#FF99FF>เซิร์ฟเวอร์ถึงหมุดหมายใหม่แล้ว: Kafra Staff เปิดให้บริการ!</color>", "", true);
	StopTimer();
OnClick:
	FocusNpc();
	ShowSprite("kafra_01", right);
	Dialog(name, "ยินดีต้อนรับสู่ Kafra Corporation~ บริการ Kafra พร้อมดูแลคุณอยู่เสมอค่ะ วันนี้ให้ช่วยอะไรดีคะ?");
	//Always offered, and every refusal below is spoken. An entry that disappears when it
	//does not apply cannot be told apart from one that was never written, and it shifts
	//what Cancel is worth underneath it.
	Option("บันทึกจุดเกิด", "ใช้คลังเก็บของ", "บริการวาร์ป", "เช่า Push Cart หรือเรียกเหยี่ยว", "ยกเลิก");
	switch(Result) 	
	{
	case 0:
		SaveSpawnPoint("South of Prontera");
		Dialog(name, "บันทึกจุดเกิดใหม่ของคุณไว้ทางใต้ของ Prontera แล้วค่ะ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
		return;
	case 1:
		if(JobId == 0 && JobLevel < 5)
		{
			Dialog(name, "ขออภัยค่ะ คลังเก็บของ Kafra เปิดให้เฉพาะนักผจญภัยที่ job level 5 ขึ้นไปเท่านั้น กลับมาใหม่ตอนเลเวลสูงกว่านี้นะคะ");
			return;
		}
		ReleaseFocus();
		OpenStorage();
		return;
	case 2:
		Dialog(name, "เลือกจุดหมายปลายทางได้เลยค่ะ");
		Option("Izlude", "Geffen", "Payon", "Morroc", "Orc Dungeon", "Alberta", "ยกเลิก" );
		switch(Result)
		{
			case 0: @TeleportCase("izlude", 91, 105, 600);
			case 1: @TeleportCase("geffen", 120, 39, 1200);
			case 2: @TeleportCase("payon", 161, 58, 1200);
			case 3: @TeleportCase("morocc", 156, 46, 1200);
			case 4: @TeleportCase("gef_fild10", 52, 326, 1700);
			case 5: @TeleportCase("alberta", 117, 56, 1000);
		}
	case 3:
		//The falcon is not rented the way a cart is - Falcon Mastery grants it, and
		//dismissing your followers is the only thing that takes it away. This is the way
		//back from that, which otherwise means logging out and in again.
		if(HasJobType("Hunter"))
		{
			if(!HasLearnedSkill(FalconMastery, 1))
			{
				Dialog(name, "เหยี่ยวจะเชื่อฟังเฉพาะ Hunter ที่เรียนวิธีเรียกมันมาแล้วเท่านั้นค่ะ ไปเรียน Falcon Mastery สิคะ แล้วมันจะบินมาหาคุณเอง~");
				return;
			}

			if(HasBird)
			{
				Dialog(name, "เหยี่ยวของคุณเกาะอยู่บนแขนนั่นไงคะ~ ดูแลมันดี ๆ นะคะ");
				return;
			}

			Dialog(name, "เหยี่ยวหายไปเหรอคะ? จ่าย 200 zeny แล้วเดี๋ยวผิวปากเรียกมันกลับมาให้ เอาไหมคะ?");
			Option("ตกลง", "ไม่ล่ะ");
			switch(Result)
			{
			case 0:
				if(GetZeny() < 200)
				{
					Dialog(name, "แหม เงินไม่พอนี่คะ กลับมาใหม่ตอนมีอย่างน้อย 200 zeny นะคะ");
					return;
				}
				DropZeny(200);
				GiveFalcon();
				Dialog(name, "มาแล้วค่ะ~ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
				return;
			}
			return;
		}

		if(!HasJobType("Merchant"))
		{
			Dialog(name, "ขออภัยค่ะ~ Push Cart มีไว้ให้ Merchant ใช้ทำการค้า ส่วนเหยี่ยวเชื่อฟังแต่ Hunter เท่านั้น กลับมาใหม่ตอนคุณเลือกเดินสายใดสายหนึ่งแล้วนะคะ");
			return;
		}

		if(HasCart)
		{
			Dialog(name, "คุณมี Push Cart อยู่แล้วนี่คะ ดูแลมันดี ๆ นะคะ~");
			return;
		}

		if(!HasLearnedSkill(PushCart, 1))
		{
			Dialog(name, "Push Cart จะไม่มีประโยชน์เลยถ้าคุณยังเข็นไม่เป็นค่ะ~ กลับมาใหม่ตอนเรียนสกิล Push Cart แล้วนะคะ");
			return;
		}

		Dialog(name, "ค่าเช่า Push Cart อยู่ที่ 850 zeny ค่ะ ตกลงไหมคะ?");
		Option("ตกลง", "ไม่ล่ะ");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 850)
			{
				Dialog(name, "แหม เงินไม่พอนี่คะ กลับมาใหม่ตอนมีอย่างน้อย 850 zeny นะคะ");
				return;
			}
			DropZeny(850);
			EquipPushCart();
		}
	}
}
