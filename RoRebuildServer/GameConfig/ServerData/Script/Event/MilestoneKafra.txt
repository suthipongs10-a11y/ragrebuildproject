#event "MilestoneEvent" //this disables this file if MilestoneEvent is not set in OperationConfig section of appsettings

macro TeleportCase(%map, %x, %y, %zeny)
{
	if(GetZeny() < %zeny)
	{
		//Dialog(name, "Sorry, you currently don't have enough zeny to use this teleport.");
		//return;
	}
	//DropZeny(%zeny);
	MoveTo(%map, %x, %y);
	return;
}

Npc("prt_fild08", "Kafra Staff", "4_F_KAFRA1", 158, 362, S) 
{
	RegisterAsKafra();
	RegisterGlobalSignal("MilestoneKafra");
	//present from the first boot rather than unlocked at level 10 - see the note on
	//EventRevealedNpc in MilestoneEvent.txt
	localstr name = "Kafra Defolty - Kafra Employee";
OnSignal:
	if(Signal == "RevealOnServerUp") ShowNpc();	else StartTimer(1s);
OnTimer 20s:
	ShowNpc();
	PlayEffectAtMyLocation("PuffOfSmoke");
OnTimer 22s:
	ServerWideMessage("<color=#FF99FF>A server milestone has been reached: the Kafra Staff!</color>", "", true);
	StopTimer();
OnClick:
	FocusNpc();
	ShowSprite("kafra_01", right);
	Dialog(name, "Welcome to the Kafra Corporation~ The Kafra Services are always here to support you. So how can I be of service today?");
	//Always offered, and every refusal below is spoken. An entry that disappears when it
	//does not apply cannot be told apart from one that was never written, and it shifts
	//what Cancel is worth underneath it.
	Option("Save", "Use Storage", "Teleport Service", "Rent a Cart or Call a Falcon", "Cancel");
	switch(Result) 	
	{
	case 0:
		SaveSpawnPoint("South of Prontera");
		Dialog(name, "Your Respawn Point has been saved south of Prontera. Thank you for using the Kafra Service.");
		return;
	case 1:
		if(JobId == 0 && JobLevel < 5)
		{
			Dialog(name, "Sorry, only adventurers who have reached job level 5 are allowed to access Kafra storage. Please come back when you have a few more levels.");
			return;
		}
		ReleaseFocus();
		OpenStorage();
		return;
	case 2:
		Dialog(name, "Please choose your destination.");
		Option("Izlude", "Geffen", "Payon", "Morroc", "Orc Dungeon", "Alberta", "Cancel" );
		switch(Result)
		{
			case 0: @TeleportCase("izlude", 91, 105, 600);
			case 1: @TeleportCase("geffen", 120, 39, 1200);
			case 2: @TeleportCase("payon", 161, 58, 1200);
			case 3: @TeleportCase("morocc", 156, 46, 1200);
			case 4: @TeleportCase("gef_fild10", 52, 326, 1700);
			case 5: @TeleportCase("alberta", 117, 56, 1000);
		}
	case 3:
		//The falcon is not rented the way a cart is - Falcon Mastery grants it, and
		//dismissing your followers is the only thing that takes it away. This is the way
		//back from that, which otherwise means logging out and in again.
		if(HasJobType("Hunter"))
		{
			if(!HasLearnedSkill(FalconMastery, 1))
			{
				Dialog(name, "A falcon will only answer a hunter who has been taught to call one. Learn Falcon Mastery and one will come to you on its own~");
				return;
			}

			if(HasBird)
			{
				Dialog(name, "Your falcon is right there on your arm~ Do take care of it.");
				return;
			}

			Dialog(name, "Lost your falcon? I can whistle it back for 200 zeny. Shall I?");
			Option("Yes", "No");
			switch(Result)
			{
			case 0:
				if(GetZeny() < 200)
				{
					Dialog(name, "Oh dear, you don't have enough money. Please come back when you have at least 200 zeny.");
					return;
				}
				DropZeny(200);
				GiveFalcon();
				Dialog(name, "There it is~ Thank you for using the Kafra Service.");
				return;
			}
			return;
		}

		if(!HasJobType("Merchant"))
		{
			Dialog(name, "I am sorry~ Push carts are for merchants and their trades, and a falcon answers to no one but a hunter. Come back when you have taken one of those paths.");
			return;
		}

		if(HasCart)
		{
			Dialog(name, "You already have a push cart. Do take care of it~");
			return;
		}

		if(!HasLearnedSkill(PushCart, 1))
		{
			Dialog(name, "A push cart is no use to you until you have learned to push one~ Come back once you know the Push Cart skill.");
			return;
		}

		Dialog(name, "The fee to rent a push cart is 850 zeny. Is that alright?");
		Option("Yes", "No");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 850)
			{
				Dialog(name, "Oh dear, you don't have enough money. Please come back when you have at least 850 zeny.");
				return;
			}
			DropZeny(850);
			EquipPushCart();
		}
	}
}
