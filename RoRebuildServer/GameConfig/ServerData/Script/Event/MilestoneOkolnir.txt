Npc("prt_fild08", "Valkyrie", "4_F_VALKYRIE", 133, 340, SE)
{
	local okolnir = 0;
	RegisterGlobalSignal("MilestoneValkyrie");
	//She stands in the camp from the first boot like everyone else. The dungeon behind
	//her is a separate question and the answer to that one is still no - see OnClick.
OnSignal:
	if(Signal == "RevealOnServerUp")
		ShowNpc();
	if(Signal == "Open")
		okolnir = 2;
	if(Signal == "Close")
	{
		okolnir = 3;
		SignalNpc("#okolnirentrance", "Close"); //close signal will come from global handler, but we'll send this signal locally
	}
	if(Signal == "ReOpen")
	{
		okolnir = 0;
	}
	if(Signal == "Reveal")
		StartTimer(1s);
OnTimer 20s:
	ShowNpc();
	PlayEffectAtMyLocation("PuffOfSmoke");
OnTimer 22s:
	ServerWideMessage("<color=#FF99FF>เซิร์ฟเวอร์บรรลุหมุดหมายใหม่แล้ว: Valkyrie!</color>", "", true);
	StopTimer();
OnClick:
	FocusNpc();
	string openOption = "";

	if(okolnir == 1)
	{
		Dialog(Name, "ประตูสู่ Okolnir กำลังจะเปิดแล้ว รอสักครู่นะ เมื่อประตูมิติเปิดแล้ว ผู้เล่นทุกคนจะเข้าได้อย่างอิสระจนกว่าจะเคลียร์สำเร็จ");
		return;
	}
	if(okolnir == 2)
	{
		Dialog(Name, "ประตูสู่ Okolnir เปิดแล้ว ขอให้โชคดี");
	}
	if(okolnir == 3)
	{
		Dialog(Name, "ยินดีด้วยที่เคลียร์ดันเจี้ยนท้าทาย Okolnir สำเร็จ หลังเคลียร์แล้วต้องรออีกประมาณ 5 นาทีจึงจะลองใหม่ได้");
		return;
	}
	if(okolnir == 0)
	{
		//Okolnir stays shut for this round of the server, so the option to open it is not
		//offered at all. This is the only place that sends "Open" to #okolnirentrance, and
		//that npc keeps its own HideNpc, so the portal cannot appear by any other route.
		//
		//Leaving openOption empty is exactly the state a player without three keys was
		//already in, so nothing below this changes. To open the dungeon again, restore the
		//key check that was here:
		//
		//  if(GetItemCount("Okolnir_Key") >= 3)
		//  {
		//      Dialog(Name, "You have the three keys required to open the door to Okolnir. Are you interested in opening the portal?");
		//      openOption = "Open the Portal";
		//  }
		Dialog(Name, "ยินดีต้อนรับนักผจญภัย ดินแดน Okolnir ถูกผนึกไว้ชั่วคราว ต่อให้มีกุญแจครบสามดอกก็ไขไม่ออกในตอนนี้ เมื่อใดมีข่าวว่าทางเปิดแล้ว ค่อยมาหาข้าอีกครั้งนะ");
	}

	Option(openOption, "อธิบายเรื่อง Okolnir", "ยกเลิก")
	switch(result)
	{
		case 0:
			Dialog(Name, "ดันเจี้ยนท้าทาย Okolnir ไม่ใช่เรื่องง่าย และไม่จำกัดจำนวนผู้เข้าร่วม ถ้าอยากรอให้มีผู้เล่นพร้อมมากกว่านี้ก่อนค่อยเข้า ก็น่าจะดีกว่านะ");
			Dialog(Name, "ถ้าเลือกดำเนินการต่อ ผู้เล่นที่ออนไลน์อยู่ทุกคนจะได้รับแจ้ง และประตูมิติจะเปิดขึ้นตรงนี้ในอีก 5 นาที ใครก็เข้าได้ และประตูจะเปิดค้างไว้จนกว่าภารกิจสุดท้ายจะสำเร็จ");
			Option("ดำเนินการต่อ", "ยกเลิก");
			if(result == 0)
			{
				TakeItem("Okolnir_Key", 3);
				ServerWideMessage("<color=#FF99FF>ผู้เล่น " + PlayerName + " กำลังเปิดประตูมิติสู่ Okolnir!</color>", "", true);
				SignalNpc("#okolnirentrance", "Open");
				okolnir = 1;
			}
		case 1:
			Dialog(Name, "Okolnir คืออีเวนต์ท้าทายที่ผู้เล่นทุกคนเข้าร่วมได้ ต้องมีผู้เล่นคนหนึ่งถือกุญแจ Okolnir 3 ดอกเพื่อเปิดประตู");
			Dialog(Name, "ประตูจะเปิดค้างไว้จนกว่าอีเวนต์จะถูกเคลียร์ หรือจนกว่าเซิร์ฟเวอร์จะปิด ตอนเปิดประตูจะมีการแจ้งเตือนล่วงหน้า 5 นาทีถึงผู้เล่นทุกคน เพื่อให้รวมตัวกันมาร่วมได้ทัน");
	}
}

Npc("que_qsch01", "#CenterAngelStage", "CLEAR_NPC", 5, 5, SE, 1, 1)
{
	HideFromView();
	local step = 0;
OnSignal:
	if(Signal == "Start")
	{
		SummonMobs(6, "OBSERVATION", 132, 134, 16, 14);
		SummonMobs(6, "RETRIBUTION", 132, 134, 16, 14);

		ServerLog("Center area event has activated.");
		step = 1;
	}

	if(Signal == "NextStep" && step == 1)
	{
		ServerLog("Angels killed, starting valkyrie phase.");
		step = 2;
		return;
	}

	if(Signal == "Stop")
	{
		step = 0;
		KillMyMobs();
		StopTimer();
	}

	StartTimer(5s); //we do both timer and on kill event, you know, just in case
OnTimer 20s:
	if(MobCount == 0)
	{
        SignalNpc("#CenterStageValkyrie", "Start");
		StopTimer();
	}
	ResetTimer();
OnMobKill:
	ServerLog("Center area mob killed. Current active: " + MobCount);
    if(MobCount == 0)
	{
        SignalNpc("#CenterStageValkyrie", "Start");
		StopTimer();
	}
}

Npc("que_qsch01", "#FinalValkyrieStage", "CLEAR_NPC", 132, 135, SE, 1, 1)
{
	HideFromView();
OnSignal:
	if(Signal == "Start")
	{
		RegisterMapWideEffect("OkolnirUnsafeZone");
		MapWideMessage("พื้นที่รอบแท่นกลางกลายเป็นเขตอันตรายแล้ว!");
		StartTimer(1s);
	}
	
	if(Signal == "End")
	{
		UnregisterImportant();
		KillMyMobs();
		EndAllEvents();
	}

	if(Signal == "Fail")
	{
		ServerLog("A full wipe of the randgris phase has occured.");
	}
OnTimer 2s:
	CreateEvent("OkolnirDamageZone", 5, 5, "", false);
OnTimer 8s:
	//activate dangerzone
	MapWideMessage("Valkyrie Randgris เข้าร่วมการต่อสู้แล้ว!");
OnTimer 10s:
	DescendSummonMonster("RANDGRIS", 132, 135, 0, 0, S, "EVENT_RANDGRIS");
	MoveLockAllMyMonsters(%(132, 136, 16, 16));
	StopTimer();
OnMobKill:
	ServerLog("Center final stage mob killed. Current active: " + MobCount);
    if(MobCount == 0)
	{
        //SignalNpc("#CenterAngelStage", "NextStep");
		StopTimer();
		EndAllEvents();
		UnregisterImportant();
		SignalNpc("#warphome", "Open");
		SignalNpc("#ReopenTimer", "Start");
		SendGlobalSignal("MilestoneValkyrie", "Close");
		if(GetGlobalInt("OkolnirFirstClear") > 1)
			MapWideMessage("อีเวนต์ Okolnir ถูกเคลียร์แล้ว!");
	}
}

AltSkillHandler("EVENT_RANDGRIS") {
	OnInit:
		FlagAsMvp();
		Cast(NoCast, 10, 2s, 5s, EasyInterrupt | HideCastBar | HideSkillName); //force her to not aggro for the first 2s on spawn in
	StateIdle:
		TryCast(Heal, 9999, 100%, 0s, 3s)[HpPercent < 50];
	StateChase:
		TryCast(CallMinion, 1, 100%, 1.5s, 0s, NoInterrupt)[HpPercent < 85 && IsNamedEventOffCooldown("SummonMinions1")] -> {
			SignalNpc("#RandgrisHelper", "Spawn");
			SetEventCooldown("SummonMinions1", -1);
		}
		TryCast(CallMinion, 1, 100%, 1.5s, 0s, NoInterrupt)[HpPercent < 45 && IsNamedEventOffCooldown("SummonMinions2")] -> {
			SignalNpc("#RandgrisHelper2", "Spawn");
			SetEventCooldown("SummonMinions2", -1);
		}
		TryCast(EarthShaker, 1, 20%, 4s, 20s, NeverInterrupt | EventOnStartCast)[HpPercent < 95]  -> {
			CreateEventOnTarget("EarthShakerCastEvent", 2 + (100 - HpPercent) / 33);
			if(HpPercent < 50)
				SetStateFlag(0, 1);
		}
		TryCast(NoCast, 1, 100%, 0, 0)[GetStateFlag(0) > 0] -> {
			CreateEventOnTarget("EarthShakerCastEvent", 2 + (100 - HpPercent) / 33);
			SetStateFlag(0, 0);
		}
		TryCast(RollingThunder, 1, 20%, 2.2s, 25s, NeverInterrupt | EventOnStartCast)[HpPercent < 90]  -> CreateEventOnTarget("ExaflareControlEvent", 2 + (100 - HpPercent) / 33);
		TryCast(LordOfVermilion, 11, 10%, 2.8s, 25s, RandomTarget);
		TryCast(JupitelThunder, 9, 5%, 0.7s, 15s);
	StateAttacking:
		TryCast(CallMinion, 1, 100%, 1.5s, 0s, NoInterrupt)[HpPercent < 80 && IsNamedEventOffCooldown("SummonMinions1")] -> {
			SignalNpc("#RandgrisHelper", "Spawn");
			SetEventCooldown("SummonMinions1", -1);
		}
		TryCast(CallMinion, 1, 100%, 1.5s, 0s, NoInterrupt)[HpPercent < 35 && IsNamedEventOffCooldown("SummonMinions2")] -> {
			SignalNpc("#RandgrisHelper2", "Spawn");
			SetEventCooldown("SummonMinions2", -1);
		}
		TryCast(EarthShaker, 1, 20%, 4s, 20s, NeverInterrupt | EventOnStartCast)[HpPercent < 95]  -> {
			CreateEventOnTarget("EarthShakerCastEvent", 2 + (100 - HpPercent) / 33);
			if(HpPercent < 50)
				SetStateFlag(0, 1);
		}
		TryCast(NoCast, 1, 100%, 0, 0)[GetStateFlag(0) > 0] -> {
			CreateEventOnTarget("EarthShakerCastEvent", 2 + (100 - HpPercent) / 33);
			SetStateFlag(0, 0);
		}
		TryCast(RollingThunder, 1, 20%, 2.2s, 25s, NeverInterrupt | EventOnStartCast)[HpPercent < 90]  -> CreateEventOnTarget("ExaflareControlEvent", 2 + (100 - HpPercent) / 33);
		TryCast(LordOfVermilion, 11, 5%, 2.8s, 25s, RandomTarget);
		TryCast(SonicBlow, 1, 5%, 1s, 10s);
		TryCast(JupitelThunder, 9, 5%, 0.7s, 15s);
		TryCast(TwoHandQuicken, 1, 100%, 0, 120s)[HpPercent < 25];
		TryCast(PowerUp, 2, 100%, 0.7s, 25s)[HpPercent < 25];
	OnDie:
		CallSpecialMonsterEvent("OkolnirEnd");
		SignalNpc("#RandgrisHelper", "Die");
		SignalNpc("#RandgrisHelper2", "Die");
}
