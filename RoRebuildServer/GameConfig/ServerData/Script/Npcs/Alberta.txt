    FocusNpc();
    Dialog(name, "เฮอะ... ข้าดูออกจากแววตาเจ้าเลยล่ะ...\nเจ้ากำลังสนใจ Turtle Island อยู่ใช่ไหม ไม่รู้ว่าไปได้ยินมาจากไหน แต่ข้าพาเจ้าไปได้นะ");

    {
        case 0:
            if(GetZeny() < 5000)
            {
                Dialog(name, "อืม... ขอโทษที เจ้ามี zeny ไม่พอ หวังว่าเจ้าจะเข้าใจนะ ฝีมือกับประสบการณ์ของข้าก็มีราคาของมัน...");
                return;

            MoveTo("tur_dun01", 157, 39);

        case 1:
            Dialog(name, "งั้นก็ได้...\nถ้าวันไหนใจอยากออกผจญภัยขึ้นมา ข้าจะรออยู่ตรงนี้แหละ");

Npc("alberta", "Fisk", "4W_SAILOR", 189, 151, SE) { 
OnClick:
    FocusNpc();
    Dialog(Name, "เฮ้ย เพื่อนยาก อยากไปไหนล่ะ?");
    Option("ไป Sunken Ship -> 250 zeny.", "ไปท่าเรือ Izlude -> 500 zeny.", "ไม่ล่ะ ขอบใจ");
    switch(Result)
    {
        case 0:
            if(GetZeny() < 250)
            {
                Dialog(Name, "เฮ้ย อย่ามาตุกติกนะ! ข้าบอกแล้วว่า 250 zeny!");
                return;
            }
            DropZeny(250);
            MoveTo("alb2trea", 43, 53);
        case 1:
            if(GetZeny() < 500)
            {
                Dialog(Name, "ไม่มี 500 zeny มาก่อน ก็อย่าหวังว่าจะได้ไป!");
                return;
            }
            DropZeny(500);
            MoveTo("izlude", 176, 182);
        case 2:
            Dialog(Name, "ก็ได้... ไอ้หนอนบก");
    }
}
