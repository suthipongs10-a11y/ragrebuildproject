Npc("alde_dun03", "Doorkeeper", "4W_F_01", 264, 16, S) {
    localstr name = "Doorkeeper Jaye"
OnClick:
    FocusNpc();
    Dialog(name, "หวังว่าคุณจะเพลิดเพลินกับหอนาฬิกา Kinaichi Rancario สถานที่ขึ้นชื่อของ Al de Baran นะ! แต่ชั้นใต้ดินที่ 4 ไม่อนุญาตให้ผู้มาเยือนเข้าไป ต้องขอให้กลับไปก่อนนะ");
    Option("เรื่องหอนาฬิกา", "เรื่องชั้น B4F", "เข้าชั้น B4F", "ยกเลิก");
    switch(Result)
    {
    case 0:
        Dialog(name, "ตำนานเล่าว่านานมาแล้ว สามยอดนักเล่นแร่แปรธาตุได้ร่วมกันก่อตั้งเมือง Al de Baran ขึ้นมา เพื่อสืบทอดศาสตร์แห่งการเล่นแร่แปรธาตุเอาไว้");
        Dialog(name, "หอคอยแห่งนี้ถูกสร้างขึ้นเพื่อเป็นคลังเก็บความรู้ โดยผสานเวทมนตร์ การเล่นแร่แปรธาตุ และวิทยาการชั้นสูงเข้าไว้ด้วยกัน");
        Dialog(name, "น่าเสียดายที่กาลเวลาผ่านไป เราสูญเสียวิธีควบคุมหอคอยและเหล่าหุ่นยนต์ที่ดูแลมันไป เพราะฉะนั้นอย่าลืมว่าที่นี่อันตรายแค่ไหนนะ");
        break;
    case 1:
        Dialog(name, "ชั้นใต้ดินลึกที่สุดใช้เก็บทั้งอุปกรณ์เวทมนตร์ อาวุธ และเป็นที่ทิ้งวัตถุดิบ เพื่อกันพวกหัวขโมย นักเล่นแร่แปรธาตุโบราณจึงผนึกทางเข้าไว้ด้วยพลังเล่นแร่แปรธาตุอันทรงพลัง");
        Dialog(name, "ถ้าอยากเข้าไป ต้องไปหากุญแจจากหุ่นยนต์อันตรายพวกนั้นในหอคอยนี้มาก่อน");
        Dialog(name, "ตัวกุญแจเองก็ถูกร่ายด้วยการเล่นแร่แปรธาตุ ใช้ได้ครั้งเดียวแล้วพลังจะหมดไป นั่นแปลว่ากุญแจ 1 ดอก พาเข้าไปได้แค่คนเดียว");
        Dialog(name, "ถ้าคุณหากุญแจมาได้ ก็ถือว่ามีสิทธิ์ลงไปชั้นล่างสุดแล้ว บอกมาได้เลย เดี๋ยวจะช่วยเปิดประตูให้ ขอให้โชคดีนะ...");
        break;
    case 2:
        if(GetItemCount("Key_of_Underground") > 0)
        {
            Dialog(name, "อืม เป็นอย่างที่คิดไว้เลย คุณนี่พิเศษจริง ๆ มีกุญแจดอกนี้ก็มีสิทธิ์เข้าไปแล้ว เดี๋ยวเปิดประตูให้ ขอให้โชคดีนะ!");
            TakeItem("Key_of_Underground");
            MoveTo("alde_dun04", 79, 267);
            return;
        }
        Dialog(name, "ขอโทษด้วยนะ ถ้าไม่มี Key of Underground ก็ยังไม่มีสิทธิ์ผ่านประตูนี้");
        break;
    case 3:
        Dialog(name, "หุ่นยนต์ของสามยอดนักเล่นแร่แปรธาตุยังเดินเพ่นพ่านอยู่ ระวังตัวด้วยนะ แล้วก็อย่าทำลายข้าวของระหว่างอยู่ที่นี่ล่ะ ขอบคุณนะ");
        break;
    }
}
