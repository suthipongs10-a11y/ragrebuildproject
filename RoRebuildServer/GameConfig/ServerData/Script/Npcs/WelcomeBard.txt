//Two doors, and they are two entries on the menu rather than one entry with a question
//behind it, because they are not variations on each other. One can be walked through as
//often as you like; the other can be walked through exactly once.
//
//  First job   always open while the character is a novice or a first job. Stats and
//              skills come back as points and neither level moves, so what is handed
//              back is exactly what there was to spend in the first place.
//  Second job  job level 50 on a first job, and after it the character's job is fixed.
//              No way back to a first job, no way across to another second job. That is
//              how it works in the game this is a copy of, and it is the reason the two
//              are separated on the menu: one of them is a decision.
//
//Ids come from Db/Jobs.csv. 0 is Novice, 1 to 6 are the first jobs, and anything from 7
//up is a second job or a special class - so "7 or more" is the whole of the lock, and it
//is checked inside each macro rather than at the menu. A guard on the menu is a guard
//that a second way in walks straight past.
//
//The six second jobs offered here are the six the server has a skill tree for. Crusader,
//Monk, Sage, Rogue, Bard, Dancer and Alchemist have no entry in Skills/SkillTree.toml,
//so changing into one would leave the player with nothing to spend points on at all.
macro JobMasterFirstJob()
{
	ShowSprite("bard_eland01", right);

	if(JobId >= 7)
	{
		Dialog(name, "เธอขึ้นอาชีพ 2 ไปแล้วนะ อาชีพจะถูกล็อคตั้งแต่วันที่เปลี่ยน กลับไปเป็นอาชีพ 1 ไม่ได้แล้ว");
		Dialog(name, "แต่รีเซ็ตสกิลกับรีเซ็ตสเตตัสยังทำให้ได้ตลอด อยากลองจัดบิลด์ใหม่ก็บอกได้เลย");
		return;
	}

	if(JobId == 0 && JobLevel < 10)
	{
		Dialog(name, "มาเปลี่ยนอาชีพใช่ไหม? ช่วยได้แน่นอน แต่ต้องทำ Job Level ให้ถึง 10 ก่อนนะ ตอนนี้อยู่ที่ " + JobLevel + " เท่านั้น");
		return;
	}

	if(JobId == 0 && UnusedSkillPoints > 0)
	{
		Dialog(name, "Job Level 10 แล้วก็จริง แต่ยังเหลือแต้มสกิลอยู่ " + UnusedSkillPoints + " แต้ม ใช้ให้หมดก่อนนะ แล้วค่อยมาใหม่");
		return;
	}

	if(GetNpcFlag(bardFlag) > 1)
		Dialog(name, "อยากเปลี่ยนอาชีพ 1 อีกครั้งใช่ไหม? ได้เลย เปลี่ยนกี่รอบก็ได้ตราบใดที่ยังไม่ขึ้นอาชีพ 2 จะเอาอาชีพอะไรดี?");
	else
		Dialog(name, "มาเปลี่ยนอาชีพใช่ไหม? ดีเลย! คิดจะเป็นอะไรดีล่ะ?");

	Option("Swordsman", "Archer", "Mage", "Acolyte", "Thief", "Merchant", "ยกเลิก");
	if(result == 6)
	{
		Dialog(name, "เปลี่ยนใจเมื่อไหร่ก็มาหาได้นะ");
		return;
	}

	var jobChoice = result + 1;
	if(jobChoice == JobId)
	{
		Dialog(name, "เลือกได้ดีมาก! แต่เสียใจด้วย นั่นคืออาชีพที่เธอเป็นอยู่แล้ว");
		return;
	}

	//Only for somebody who has something to lose. A novice has no stats or skills of
	//their own worth resetting and the reset below skips them, so reading this list at
	//them would be warning about something that is not going to happen.
	if(JobId > 0)
		Dialog(name, "<color=#CC5500>• สกิลกับสเตตัสถูกรีเซ็ต คืนแต้มให้ครบเท่าเดิม\n" +
					 "• Base Level กับ Job Level ไม่ลด แต้มที่ได้จึงเท่าเดิมทุกแต้ม\n" +
					 "• ของที่ใส่อยู่ถูกถอดกลับเข้ากระเป๋า รถเข็นถูกเก็บ\n" +
					 "• ไอเทมกับเงินไม่หายไปไหน</color>");

	Dialog(name, "แน่ใจนะว่าจะเป็นอาชีพนี้?");
	Option("แน่ใจ", "ขอคิดดูก่อน");
	if(result == 1)
	{
		Dialog(name, "ตัดสินใจได้เมื่อไหร่ก็มาบอกได้เลย");
		return;
	}

	ReleaseFocus();
	if(JobId > 0)
	{
		StatPointReset();
		SkillReset();
	}
	ChangePlayerJob(jobChoice);
	ShowEffectOnPlayer("JobUp");

	ShowSprite("bard_eland02", right);
	if(GetNpcFlag(bardFlag) == 1)
	{
		GiveZeny(2000);
		GiveItem("Novice_Potion", 60);
		GiveItem("Fly_Wing", 15);
		GiveItem("Butterfly_Wing", 5);
		GiveItem("Concentration_Potion", 2);

		if(jobChoice == 1) GiveItem("Sword");
		if(jobChoice == 2) { GiveItem("Arrow", 500); GiveItem("Bow"); }
		if(jobChoice == 3) GiveItem("Novice_Rod");
		if(jobChoice == 4) GiveItem("Mace");
		if(jobChoice == 5) GiveItem("Cutter");
		if(jobChoice == 6) GiveItem("Axe");

		SetNpcFlag(bardFlag, 2);
		Dialog(name, "เรียบร้อย! แถมของให้เล็กน้อยไว้ใช้ตอนเริ่มต้น ถ้าเปลี่ยนใจอยากย้ายอาชีพอีกก็มาบอกได้");
	}
	else
	{
		Dialog(name, "เรียบร้อย! ต้องการเปลี่ยนอีกเมื่อไหร่ก็มาหาได้เลย");
	}

	return;
}

macro JobMasterSecondJob()
{
	ShowSprite("bard_eland01", right);

	if(JobId >= 7)
	{
		Dialog(name, "เธอเป็นอาชีพ 2 อยู่แล้ว และอาชีพ 2 เปลี่ยนไม่ได้อีกแล้วนะ ล็อคตั้งแต่วันที่เปลี่ยนเลย");
		return;
	}

	if(JobId == 0)
	{
		Dialog(name, "ใจเย็นก่อน! ต้องขึ้นอาชีพ 1 ให้ได้ก่อน แล้วทำ Job Level ให้ครบ 50 ค่อยมาคุยเรื่องอาชีพ 2 กัน");
		return;
	}

	if(JobLevel < 50)
	{
		Dialog(name, "อยากไปอาชีพ 2 ใช่ไหม? ต้องทำ Job Level ให้ครบ 50 ก่อนนะ ตอนนี้เธออยู่ที่ " + JobLevel + " เท่านั้น");
		return;
	}

	if(UnusedSkillPoints > 0)
	{
		Dialog(name, "Job Level 50 แล้วก็จริง แต่ยังเหลือแต้มสกิลอยู่ " + UnusedSkillPoints + " แต้ม " +
					 "ใช้ให้หมดก่อนนะ แล้วค่อยมาใหม่");
		return;
	}

	//The second job that follows from what they are now. Anything else has no skill tree
	//on this server, so it is not offered at all rather than offered and empty.
	//
	//The name is printed per branch rather than held in a variable, because a variable in
	//this script language is an int and nothing else — a var holding a string compiles to
	//"cannot convert string to int", which is exactly what it did.
	var target = 0;
	if(JobId == 1) target = 7;
	if(JobId == 2) target = 10;
	if(JobId == 3) target = 8;
	if(JobId == 4) target = 9;
	if(JobId == 5) target = 11;
	if(JobId == 6) target = 12;

	if(target == 0)
	{
		Dialog(name, "อาชีพของเธอยังไม่มีสายอาชีพ 2 ให้ไปต่อในตอนนี้");
		return;
	}

	if(target == 7)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Knight ได้เลย");
	if(target == 10) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Hunter ได้เลย");
	if(target == 8)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Wizard ได้เลย");
	if(target == 9)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Priest ได้เลย");
	if(target == 11) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Assassin ได้เลย");
	if(target == 12) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Blacksmith ได้เลย");
	Dialog(name, "<color=#CC0000>อ่านให้ดีก่อนนะ การเป็นอาชีพ 2 คือการเริ่มใหม่ ไม่ใช่การอัปเกรด</color>");
	Dialog(name, "<color=#CC5500>• <b>Base Level จะถูกลดเหลือ 10</b> (ตอนนี้เธออยู่เลเวล " + Level + ")\n" +
				 "• Job Level กลับไปเริ่มที่ 1\n" +
				 "• สกิลและสเตตัสถูกรีเซ็ตทั้งหมด คืนแต้มให้ตามเลเวลใหม่\n" +
				 "• ของที่ใส่อยู่ถูกถอดกลับเข้ากระเป๋า รถเข็นถูกเก็บ\n" +
				 "• <b>อาชีพจะถูกล็อคถาวร เปลี่ยนอาชีพไม่ได้อีกเลย</b>\n" +
				 "• ไอเทมกับเงินไม่หายไปไหน</color>");
	Dialog(name, "<color=#CC0000>ย้อนกลับไม่ได้นะ</color> ตราบใดที่ยังเป็นอาชีพ 1 เธอจะสลับไปมากี่รอบก็ได้ " +
				 "แต่พอขึ้นอาชีพ 2 แล้วคือจบ จะเปลี่ยนอาชีพเลยไหม?");

	Option("เอาเลย", "ขอคิดดูก่อน");
	if(result == 1)
	{
		Dialog(name, "ไม่ต้องรีบ ตัดสินใจได้แล้วค่อยมา");
		return;
	}

	ReleaseFocus();
	//given back before the level drops, or the character is left owing points: what a
	//character has spare is their level's allowance less what is spent, and at a lower
	//level than they spent at, that subtraction goes negative
	StatPointReset();
	SkillReset();
	ChangePlayerJob(target);
	//ChangeJob only puts the job level back when leaving novice, so this step says it
	SetJobLevel(1);
	SetBaseLevel(10);
	ShowEffectOnPlayer("JobUp");

	ShowSprite("bard_eland02", right);
	Dialog(name, "เรียบร้อย! ยินดีด้วยนะ ไปลุยได้เลย");
	return;
}

macro BardRandomizeAppearance()
{
	ShowSprite("bard_eland01", right);
	Dialog(name, "อยากสุ่มรูปลักษณ์ใหม่สินะ? ได้เลย จะเปลี่ยนตรงไหนดี?");
	Option("เปลี่ยนเพศ", "สุ่มทรงผม", "สุ่มสีผม", "ยกเลิก");

	switch(result)
	{
		case 0: ReleaseFocus(); ChangePlayerGender();
		case 1: ReleaseFocus(); ChangePlayerHairToRandom();
		case 2: ReleaseFocus(); ChangePlayerHairColorToRandom();
		case 3: 
			Dialog(name, "ไม่ต้องรีบ เปลี่ยนใจเมื่อไหร่ก็มาบอกได้");
			return;
	}

	ShowSprite("bard_eland02", right);
	Dialog(name, "เรียบร้อย! ขอให้ชอบลุคใหม่นะ อยากเปลี่ยนอีกเมื่อไหร่ก็มาบอกได้เลย");
	return;
}

macro BardWarps()
{
	ShowSprite("bard_eland01", right);
	Dialog(name, "เธอกดปุ่ม warp ตรงมุมขวาล่างไปที่ไหนก็ได้อยู่แล้ว แต่อยากให้ข้าแนะนำที่เที่ยวสักหน่อยไหม?");
	Option("ช่วยแนะนำที่ให้หน่อย", "ขอเลือกเอง", "ยกเลิก");
	if(result == 2) return;
	var range = 0;
	if(result == 0)
	{
		Dialog(name, "ดูซิ เธอเลเวล " + Level + " ใช่ไหม? ลองแมพพวกนี้เป็นไง?");
		if(Level < 15) range = 0;
		else if(Level < 30) range = 1;
		else if(Level < 45) range = 2;
		else if(Level < 60) range = 3;
		else if(Level < 70) range = 4;
		else if(Level < 80) range = 5;
		else range = 6;
	}
	else
	{
		Option("เลเวล 1-15", "เลเวล 16-30", "เลเวล 31-45", "เลเวล 46-60", "ยากที่สุดเท่าที่มี", "ยกเลิก");
		if(result > 6) return;
		range = result;
	}
	
	if(range == 0) @BardWarps1to10();
	if(range == 1) @BardWarps11to20();
	if(range == 2) @BardWarps21to30();
	if(range == 3) @BardWarps31to40();
	if(range > 3) @BardWarp41AndAbove();
	return;
}

Npc("prt_fild08", "Class Master", "1_M_BARD", 153, 357, SE, 1, 1)  //175, 374, SW
{
	localstr bardFlag = "PrtBardStatus";
	localstr name = "Class Master"
	OnClick:
		FocusNpc();
		ShowSprite("bard_eland01", right);

		if(GetNpcFlag(bardFlag) == 0)
		{
			Dialog(name, "สวัสดี ยินดีต้อนรับ! ข้าคือ <color=#0066CC>Class Master</color> มีหน้าที่เปลี่ยนอาชีพให้นักผจญภัยอย่างเธอ " +
						 "ทั้งอาชีพ 1 และอาชีพ 2 รวมถึงรีเซ็ตสกิลกับสเตตัสให้ด้วย");
			Dialog(name, "<color=#CC5500>จำกฎข้อเดียวนี้ไว้ให้ดี</color> — <b>อาชีพ 1 เปลี่ยนกี่รอบก็ได้</b> " +
						 "อยากลองเป็นอะไรก็มาบอกข้า สกิลกับสเตตัสจะถูกรีเซ็ตคืนแต้มให้ครบ เลเวลไม่ลดสักหน่อย");
			Dialog(name, "<color=#CC0000>แต่วันที่เธอขึ้นอาชีพ 2 คือวันที่อาชีพถูกล็อค</color> " +
						 "หลังจากนั้นเปลี่ยนอาชีพไม่ได้อีกเลย ค่อย ๆ คิดให้ดีก่อนนะ");
			Dialog(name, "มาครั้งแรกสินะ เดี๋ยวข้าให้ของติดตัวไปหน่อย");
			SetNpcFlag(bardFlag, 1);
			GiveZeny(1000);
			GiveItem("Novice_Potion", 30);
			Dialog(name, "เอาล่ะ ให้ข้าช่วยอะไรดี?");
		}
		else
		{
			Dialog(name, "กลับมาแล้วหรือ! ให้ข้าช่วยอะไรดี?");
		}

		//Both job changes are named on the menu from the first visit rather than hidden
		//behind one "change job" entry. They are not the same errand: one of them can be
		//taken back and the other cannot, and a menu that hides that behind a shared name
		//is a menu that lets somebody walk into the permanent one by accident.
		//
		//The choice is copied out before anything runs, and put back afterwards. Every
		//macro below asks a question of its own, and asking a question writes the answer
		//into result - so the loop used to be steered by whatever the last inner menu
		//happened to return, and cancelling out of the job list quietly ended the whole
		//conversation. Now the loop only ever reads what was picked here.
		var choice = 0;
		result = 0;
		while(result < 6)
		{
			Option("เกี่ยวกับเซิร์ฟเวอร์", "เปลี่ยนอาชีพ 1", "เปลี่ยนเป็นอาชีพ 2", "รีเซ็ตสกิล", "รีเซ็ตสเตตัส", "เปลี่ยนรูปลักษณ์", "ยกเลิก");
			choice = result;

			if(choice == 0) @BardAboutRebuild();
			if(choice == 1) @JobMasterFirstJob();
			if(choice == 2) @JobMasterSecondJob();
			if(choice == 3) @BardSkillReset();
			if(choice == 4) @BardStatReset();
			if(choice == 5) @BardRandomizeAppearance();

			result = choice;
		}

}
