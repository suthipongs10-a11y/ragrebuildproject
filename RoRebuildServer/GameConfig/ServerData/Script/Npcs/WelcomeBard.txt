//Novice to first job, and first job to second. Second job needs job level 50 on the
//first, which is the whole of the first job, and it puts the job level back to 10 so
//the new tree is learned rather than arriving already finished.
//
//The six second jobs offered here are the six the server has a skill tree for. Crusader,
//Monk, Sage, Rogue, Bard, Dancer and Alchemist have no entry in Skills/SkillTree.toml,
//so changing into one would leave the player with nothing to spend points on at all.
macro JobMasterFirstJob()
{
	if(GetNpcFlag(bardFlag) > 1)
		Dialog(name, "อยากเปลี่ยนอาชีพอีกครั้งใช่ไหม? ได้เลย แต่ต้องรีเซ็ตสกิลกับสเตตัสด้วยนะ จะเอาอาชีพอะไรดี?");
	else
		Dialog(name, "มาเปลี่ยนอาชีพใช่ไหม? ดีเลย! คิดจะเป็นอะไรดีล่ะ?");

	Option("Swordsman", "Archer", "Mage", "Acolyte", "Thief", "Merchant", "ยกเลิก");
	if(result == 6)
	{
		Dialog(name, "เปลี่ยนใจเมื่อไหร่ก็มาหาได้นะ");
		return;
	}

	var jobChoice = result + 1;
	if(jobChoice == JobId)
	{
		Dialog(name, "เลือกได้ดีมาก! แต่เสียใจด้วย นั่นคืออาชีพที่เธอเป็นอยู่แล้ว");
		return;
	}

	Dialog(name, "แน่ใจนะว่าจะเป็นอาชีพนี้?");
	Option("แน่ใจ", "ขอคิดดูก่อน");
	if(result == 1)
	{
		Dialog(name, "ตัดสินใจได้เมื่อไหร่ก็มาบอกได้เลย");
		return;
	}

	ReleaseFocus();
	if(JobId > 0)
	{
		StatPointReset();
		SkillReset();
	}
	ChangePlayerJob(jobChoice);
	ShowEffectOnPlayer("JobUp");

	ShowSprite("bard_eland02", right);
	if(GetNpcFlag(bardFlag) == 1)
	{
		GiveZeny(2000);
		GiveItem("Novice_Potion", 60);
		GiveItem("Fly_Wing", 15);
		GiveItem("Butterfly_Wing", 5);
		GiveItem("Concentration_Potion", 2);

		if(jobChoice == 1) GiveItem("Sword");
		if(jobChoice == 2) { GiveItem("Arrow", 500); GiveItem("Bow"); }
		if(jobChoice == 3) GiveItem("Novice_Rod");
		if(jobChoice == 4) GiveItem("Mace");
		if(jobChoice == 5) GiveItem("Cutter");
		if(jobChoice == 6) GiveItem("Axe");

		SetNpcFlag(bardFlag, 2);
		Dialog(name, "เรียบร้อย! แถมของให้เล็กน้อยไว้ใช้ตอนเริ่มต้น ถ้าเปลี่ยนใจอยากย้ายอาชีพอีกก็มาบอกได้");
	}
	else
	{
		Dialog(name, "เรียบร้อย! ต้องการเปลี่ยนอีกเมื่อไหร่ก็มาหาได้เลย");
	}

	return;
}

macro JobMasterSecondJob()
{
	if(JobLevel < 50)
	{
		Dialog(name, "อยากไปอาชีพ 2 ใช่ไหม? ต้องทำ Job Level ให้ครบ 50 ก่อนนะ ตอนนี้เธออยู่ที่ " + JobLevel + " เท่านั้น");
		return;
	}

	if(UnusedSkillPoints > 0)
	{
		Dialog(name, "Job Level 50 แล้วก็จริง แต่ยังเหลือแต้มสกิลอยู่ " + UnusedSkillPoints + " แต้ม " +
					 "ใช้ให้หมดก่อนนะ แล้วค่อยมาใหม่");
		return;
	}

	//The second job that follows from what they are now. Anything else has no skill tree
	//on this server, so it is not offered at all rather than offered and empty.
	//
	//The name is printed per branch rather than held in a variable, because a variable in
	//this script language is an int and nothing else — a var holding a string compiles to
	//"cannot convert string to int", which is exactly what it did.
	var target = 0;
	if(JobId == 1) target = 7;
	if(JobId == 2) target = 10;
	if(JobId == 3) target = 8;
	if(JobId == 4) target = 9;
	if(JobId == 5) target = 11;
	if(JobId == 6) target = 12;

	if(target == 0)
	{
		Dialog(name, "อาชีพของเธอยังไม่มีสายอาชีพ 2 ให้ไปต่อในตอนนี้");
		return;
	}

	if(target == 7)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Knight ได้เลย");
	if(target == 10) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Hunter ได้เลย");
	if(target == 8)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Wizard ได้เลย");
	if(target == 9)  Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Priest ได้เลย");
	if(target == 11) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Assassin ได้เลย");
	if(target == 12) Dialog(name, "Job Level 50 เต็ม แต้มลงครบแล้ว! เธอพร้อมเป็น Blacksmith ได้เลย");
	Dialog(name, "<color=#CC0000>อ่านให้ดีก่อนนะ การเป็นอาชีพ 2 คือการเริ่มใหม่ ไม่ใช่การอัปเกรด</color>");
	Dialog(name, "<color=#CC5500>• <b>Base Level จะถูกลดเหลือ 10</b> (ตอนนี้เธออยู่เลเวล " + Level + ")\n" +
				 "• Job Level กลับไปเริ่มที่ 1\n" +
				 "• สกิลและสเตตัสถูกรีเซ็ตทั้งหมด คืนแต้มให้ตามเลเวลใหม่\n" +
				 "• ไอเทมกับเงินไม่หายไปไหน</color>");
	Dialog(name, "<color=#CC0000>ย้อนกลับไม่ได้นะ</color> จะเปลี่ยนอาชีพเลยไหม?");

	Option("เอาเลย", "ขอคิดดูก่อน");
	if(result == 1)
	{
		Dialog(name, "ไม่ต้องรีบ ตัดสินใจได้แล้วค่อยมา");
		return;
	}

	ReleaseFocus();
	//given back before the level drops, or the character is left owing points: what a
	//character has spare is their level's allowance less what is spent, and at a lower
	//level than they spent at, that subtraction goes negative
	StatPointReset();
	SkillReset();
	ChangePlayerJob(target);
	//ChangeJob only puts the job level back when leaving novice, so this step says it
	SetJobLevel(1);
	SetBaseLevel(10);
	ShowEffectOnPlayer("JobUp");

	ShowSprite("bard_eland02", right);
	Dialog(name, "เรียบร้อย! ยินดีด้วยนะ ไปลุยได้เลย");
	return;
}

macro BardChangeJob()
{
	ShowSprite("bard_eland01", right);

	if(JobId == 0 && JobLevel < 10)
	{
		Dialog(name, "มาเปลี่ยนอาชีพใช่ไหม? ช่วยได้แน่นอน แต่ต้องทำ Job Level ให้ถึง 10 ก่อนนะ");
		return;
	}

	if(JobId == 0 && UnusedSkillPoints > 0)
	{
		Dialog(name, "Job Level 10 แล้วก็จริง แต่ต้องใช้แต้มสกิลที่เหลือให้หมดก่อน ถึงจะเปลี่ยนอาชีพให้ได้");
		return;
	}

	//a novice has only one road out, so it is not worth asking which
	if(JobId == 0)
	{
		@JobMasterFirstJob();
		return;
	}

	Option("เปลี่ยนเป็นอาชีพ 2", "เปลี่ยนอาชีพ 1 ใหม่", "ยกเลิก");
	if(result == 0)
	{
		@JobMasterSecondJob();
		return;
	}
	if(result == 1)
	{
		@JobMasterFirstJob();
		return;
	}

	Dialog(name, "ได้เลย ต้องการอะไรก็บอกได้");
	return;
}

macro BardSkillReset()
{
	ShowSprite("bard_eland01", right);
	if(JobId == 0)
	{
		Dialog(name, "Looking to reset your skills are you? Unfortunately I can't reset your novice skills so I can't help you out much there.");
		return;
	}
	else
	{
		Dialog(name, "You're looking to reset your skills are you? I can reset all of your skill points you used, except those put into your novice skills. Are you sure you want to?");
		Option("Reset Skills", "No thanks");

		if(result == 0)
		{
			SkillReset();
			Dialog(name, "It's done! Be sure to be more careful with those skill points next time.");
			return;
		}

		Dialog(name, "That's fine by me, let me know if you ever want that skill reset sometime.");

	}
}


macro BardStatReset()
{
	ShowSprite("bard_eland01", right);

		StatPointReset();
		Dialog(name, "It's done! Be sure to be more careful with those stat points next time.");
		return;
	}

	Dialog(name, "That's fine by me, let me know if you ever want that stat reset sometime.");
}


macro BardRandomizeAppearance()
{
	ShowSprite("bard_eland01", right);
	Dialog(name, "You want to randomize your appearance? Sure, what exactly do you want to change?");
	Option("Change Gender", "Randomize Hair Style", "Randomize Hair Color", "Cancel");

	switch(result)
	{
		case 0: ReleaseFocus(); ChangePlayerGender();
		case 1: ReleaseFocus(); ChangePlayerHairToRandom();
		case 2: ReleaseFocus(); ChangePlayerHairColorToRandom();
		case 3: 
			Dialog(name, "No pressure, just let me know if you change your mind.");
			return;
	}

	ShowSprite("bard_eland02", right);
	Dialog(name, "We're done! Enjoy the new you, and let me know if you ever need a change.");
	return;
}

macro FullCharacterReset()
{

		Option("<color=#FF5555>Reset my character</color>", "Cancel");

			ResetCharacterToInitialState();

macro BardWarps()
{
	ShowSprite("bard_eland01", right);
	Dialog(name, "You can use that warp button in the bottom right to go anywhere you want, but maybe you'd rather have suggestions?");
	Option("Recommend me somewhere", "I'd like to chose", "Cancel");
	if(result == 2) return;
	var range = 0;
	if(result == 0)
	{
		Dialog(name, "Lets see, you're level " + Level + ", right? How about one of these maps?");
		if(Level < 15) range = 0;
		else if(Level < 30) range = 1;
		else if(Level < 45) range = 2;
		else if(Level < 60) range = 3;
		else if(Level < 70) range = 4;
		else if(Level < 80) range = 5;
		else range = 6;
	}
	else
	{
		Option("Level 1-15", "Level 16-30", "Level 31-45", "Level 46-60", "As hard as it gets", "Cancel");
		if(result > 6) return;
		range = result;
	}
	
	if(range == 0) @BardWarps1to10();
	if(range == 1) @BardWarps11to20();
	if(range == 2) @BardWarps21to30();
	if(range == 3) @BardWarps31to40();
	if(range > 3) @BardWarp41AndAbove();
	return;
}
