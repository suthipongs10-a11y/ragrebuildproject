

macro ShipCase(%map, %x, %y, %zeny)
{
    if(GetZeny() < %zeny)
    {
        Dialog(Name, %zeny + " zeny! แค่ " + %zeny + " zeny ก็ได้นั่งเรือแล้ว!");
        return;
    }
    DropZeny(%zeny);
    MoveTo(%map, %x, %y);
    return;
}

Npc("izlude", "Sailor", "4W_SAILOR", 201, 181, W) 
{
    RegisterAsWarpNpc();
OnClick:
    FocusNpc();
    Dialog(Name, "ทุกคนโปรดทราบ! เชิญขึ้นมาโลดแล่นไปกับสายลม บนเรือที่เร็วที่สุดใน Rune-Midgard!");
    Option("เกาะ Byalan -> 150z", "ท่าเรือ Alberta -> 500z", "ยกเลิก" );
    switch(Result)
    {
        case 0: @ShipCase("izlu2dun", 107, 50, 150);
        case 1: @ShipCase("alberta", 188, 169, 500);
    }
}
