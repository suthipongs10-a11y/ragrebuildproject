
macro Kafra(%map, %x, %y, %facing, %npcName, %chatName, %sprite, %cutin, %saveLocation, %saveDesc, %teleportMacro)
{
	Npc(%map, %npcName, %sprite, %x, %y, %facing) 
	{
		RegisterAsKafra();
		localstr name = %chatName;
	OnClick:
		FocusNpc();
		ShowSprite(%cutin, right);
		Dialog(name, "ยินดีต้อนรับสู่ Kafra Corporation~ บริการ Kafra พร้อมดูแลคุณอยู่เสมอค่ะ วันนี้ให้ช่วยอะไรดีคะ?");
		//Always offered, and the refusal is spoken rather than being a menu entry that
		//quietly is not there. A service you cannot see is one you cannot tell apart from a
		//service that was never added, which is exactly how the last version read.
		Option("บันทึกจุดเกิด", "ใช้คลังเก็บของ", "บริการวาร์ป", "เช่าพาหนะหรือเรียกเหยี่ยว", "ยกเลิก");
		switch(Result)
		{
		case 0:
			SaveSpawnPoint(%saveLocation);
			Dialog(name, "บันทึกจุดเกิดใหม่ของคุณไว้" + %saveDesc + " แล้วค่ะ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
			return;
		case 1:
			if(JobId == 0 && JobLevel < 5)
			{
				Dialog(name, "ขออภัยค่ะ คลังเก็บของ Kafra เปิดให้เฉพาะนักผจญภัยที่ job level 5 ขึ้นไปเท่านั้น กลับมาใหม่ตอนเลเวลสูงกว่านี้นะคะ");
				return;
			}
			ReleaseFocus();
			OpenStorage();
			//Dialog(name, "Sorry, currently our storage service is unavailable. Please try back some other time.");
			return;
		case 2:
			Dialog(name, "เลือกจุดหมายปลายทางได้เลยค่ะ");
			@%teleportMacro();
		case 3:
			@KafraRentMount(name);
		}
	}
}

//The one place either rental is decided, so the two Kafra macros cannot drift apart.
//Everything it can refuse, it refuses out loud: a service you cannot see is one you cannot
//tell apart from a service that was never added.
//
//Written out flat, with no macro calling another. A macro's parameters are looked up in
//whichever macro is on top of the stack, one deep, so a macro cannot hand its own parameter
//to the next one - %kafraName inside a nested call would be looked for in the callee, which
//has never heard of it. Passing a plain name in from the NPC body, as here, is fine.
macro KafraRentMount(%kafraName)
{
	if(HasJobType("Merchant"))
	{
		if(HasCart)
		{
			Dialog(%kafraName, "คุณมี Push Cart อยู่แล้วนี่คะ ดูแลมันดี ๆ นะคะ~");
			return;
		}

		Dialog(%kafraName, "เช่า Push Cart ราคา 200 zeny และเป็นของคุณไปเลยค่ะ จะให้ไปเอามาให้ไหมคะ?");
		Option("ตกลง", "ไม่ล่ะ");
		switch(Result)
		{
		case 0:
			//the zeny is taken only once the thing being paid for is about to be handed
			//over, so a character who cannot afford it walks away no poorer
			if(GetZeny() < 200)
			{
				Dialog(%kafraName, "แหม เงินไม่พอนี่คะ กลับมาใหม่ตอนมีอย่างน้อย 200 zeny นะคะ");
				return;
			}
			DropZeny(200);
			EquipPushCart();
			Dialog(%kafraName, "เรียบร้อยค่ะ~ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
			return;
		}
		return;
	}

	if(HasJobType("Knight"))
	{
		if(HasPeco)
		{
			Dialog(%kafraName, "คุณขี่ Peco Peco อยู่แล้วนี่คะ ดูแลมันดี ๆ นะคะ~");
			return;
		}

		Dialog(%kafraName, "เช่า Peco Peco ราคา 500 zeny ค่ะ จะให้จูงออกมาให้ไหมคะ?");
		Option("ตกลง", "ไม่ล่ะ");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 500)
			{
				Dialog(%kafraName, "แหม เงินไม่พอนี่คะ กลับมาใหม่ตอนมีอย่างน้อย 500 zeny นะคะ");
				return;
			}
			DropZeny(500);
			GivePeco();
			Dialog(%kafraName, "ดูแลมันดี ๆ นะคะ~ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
			return;
		}
		return;
	}

	if(HasJobType("Hunter"))
	{
		//The falcon is not rented the way the other two are - Falcon Mastery grants it, and
		//dismissing your followers is the only thing that takes it away. This is the way
		//back from that, which otherwise means logging out and in again.
		if(!HasLearnedSkill(FalconMastery, 1))
		{
			Dialog(%kafraName, "เหยี่ยวจะเชื่อฟังเฉพาะ Hunter ที่เรียนวิธีเรียกมันมาแล้วเท่านั้นค่ะ ไปเรียน Falcon Mastery สิคะ แล้วมันจะบินมาหาคุณเอง~");
			return;
		}

		if(HasBird)
		{
			Dialog(%kafraName, "เหยี่ยวของคุณเกาะอยู่บนแขนนั่นไงคะ~ ดูแลมันดี ๆ นะคะ");
			return;
		}

		Dialog(%kafraName, "เหยี่ยวหายไปเหรอคะ? จ่าย 200 zeny แล้วเดี๋ยวผิวปากเรียกมันกลับมาให้ เอาไหมคะ?");
		Option("ตกลง", "ไม่ล่ะ");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 200)
			{
				Dialog(%kafraName, "แหม เงินไม่พอนี่คะ กลับมาใหม่ตอนมีอย่างน้อย 200 zeny นะคะ");
				return;
			}
			DropZeny(200);
			GiveFalcon();
			Dialog(%kafraName, "มาแล้วค่ะ~ ขอบคุณที่ใช้บริการ Kafra ค่ะ");
			return;
		}
		return;
	}

	Dialog(%kafraName, "ขออภัยค่ะ~ Push Cart มีไว้ให้ Merchant ใช้ทำการค้า Peco Peco ก็ยอมให้เฉพาะ Knight ขี่ ส่วนเหยี่ยวเชื่อฟังแต่ Hunter เท่านั้น กลับมาใหม่ตอนคุณเลือกเดินสายใดสายหนึ่งแล้วนะคะ");
	return;
}

}
