
macro Kafra(%map, %x, %y, %facing, %npcName, %chatName, %sprite, %cutin, %saveLocation, %saveDesc, %teleportMacro)
{
	Npc(%map, %npcName, %sprite, %x, %y, %facing) 
	{
		RegisterAsKafra();
		localstr name = %chatName;
	OnClick:
		FocusNpc();
		ShowSprite(%cutin, right);
		Dialog(name, "Welcome to the Kafra Corporation~ The Kafra Services are always here to support you. So how can I be of service today?");
		//Always offered, and the refusal is spoken rather than being a menu entry that
		//quietly is not there. A service you cannot see is one you cannot tell apart from a
		//service that was never added, which is exactly how the last version read.
		Option("Save", "Use Storage", "Teleport Service", "Rent a Mount or Call a Falcon", "Cancel");
		switch(Result)
		{
		case 0:
			SaveSpawnPoint(%saveLocation);
			Dialog(name, "Your Respawn Point has been saved " + %saveDesc + ". Thank you for using the Kafra Service.");
			return;
		case 1:
			if(JobId == 0 && JobLevel < 5)
			{
				Dialog(name, "Sorry, only adventurers who have reached job level 5 are allowed to access Kafra storage. Please come back when you have a few more levels.");
				return;
			}
			ReleaseFocus();
			OpenStorage();
			//Dialog(name, "Sorry, currently our storage service is unavailable. Please try back some other time.");
			return;
		case 2:
			Dialog(name, "Please choose your destination.");
			@%teleportMacro();
		case 3:
			@KafraRentMount(name);
		}
	}
}

//The one place either rental is decided, so the two Kafra macros cannot drift apart.
//Everything it can refuse, it refuses out loud: a service you cannot see is one you cannot
//tell apart from a service that was never added.
//
//Written out flat, with no macro calling another. A macro's parameters are looked up in
//whichever macro is on top of the stack, one deep, so a macro cannot hand its own parameter
//to the next one - %kafraName inside a nested call would be looked for in the callee, which
//has never heard of it. Passing a plain name in from the NPC body, as here, is fine.
macro KafraRentMount(%kafraName)
{
	if(HasJobType("Merchant"))
	{
		if(HasCart)
		{
			Dialog(%kafraName, "You already have a push cart. Do take care of it~");
			return;
		}

		Dialog(%kafraName, "A push cart is 200 zeny to rent, and it is yours to keep. Shall I fetch one?");
		Option("Yes", "No");
		switch(Result)
		{
		case 0:
			//the zeny is taken only once the thing being paid for is about to be handed
			//over, so a character who cannot afford it walks away no poorer
			if(GetZeny() < 200)
			{
				Dialog(%kafraName, "Oh dear, you don't have enough money. Please come back when you have at least 200 zeny.");
				return;
			}
			DropZeny(200);
			EquipPushCart();
			Dialog(%kafraName, "There you are~ Thank you for using the Kafra Service.");
			return;
		}
		return;
	}

	if(HasJobType("Knight"))
	{
		if(HasPeco)
		{
			Dialog(%kafraName, "You are already riding a Peco Peco. Do take care of it~");
			return;
		}

		Dialog(%kafraName, "A Peco Peco is 500 zeny to rent. Shall I bring one out?");
		Option("Yes", "No");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 500)
			{
				Dialog(%kafraName, "Oh dear, you don't have enough money. Please come back when you have at least 500 zeny.");
				return;
			}
			DropZeny(500);
			GivePeco();
			Dialog(%kafraName, "Do take care of it~ Thank you for using the Kafra Service.");
			return;
		}
		return;
	}

	if(HasJobType("Hunter"))
	{
		//The falcon is not rented the way the other two are - Falcon Mastery grants it, and
		//dismissing your followers is the only thing that takes it away. This is the way
		//back from that, which otherwise means logging out and in again.
		if(!HasLearnedSkill(FalconMastery, 1))
		{
			Dialog(%kafraName, "A falcon will only answer a hunter who has been taught to call one. Learn Falcon Mastery and one will come to you on its own~");
			return;
		}

		if(HasBird)
		{
			Dialog(%kafraName, "Your falcon is right there on your arm~ Do take care of it.");
			return;
		}

		Dialog(%kafraName, "Lost your falcon? I can whistle it back for 200 zeny. Shall I?");
		Option("Yes", "No");
		switch(Result)
		{
		case 0:
			if(GetZeny() < 200)
			{
				Dialog(%kafraName, "Oh dear, you don't have enough money. Please come back when you have at least 200 zeny.");
				return;
			}
			DropZeny(200);
			GiveFalcon();
			Dialog(%kafraName, "There it is~ Thank you for using the Kafra Service.");
			return;
		}
		return;
	}

	Dialog(%kafraName, "I am sorry~ Push carts are for merchants and their trades, a Peco Peco will only carry a knight, and a falcon answers to no one but a hunter. Come back when you have taken one of those paths.");
	return;
}
