//Amatsu dungeon entrance. Simplified into one NPC instead of a full quest (for now)
Npc("ama_in02", "Soldier#10", "8_M_JPNSOLDIER", 115, 177, E)
{
    localstr name = "Jyuichiro";
OnClick:
    FocusNpc();
    Dialog(name, "มาทำอะไรที่นี่? มีธุระอะไรหรือเปล่า?");
    Option( "ถามเรื่องประตู", "เข้าประตู", "กลับก่อน")
    switch(Result)
    {
        case 1:
            Dialog(name, "ถ้ารู้อยู่แล้วว่ากำลังจะเจอกับอะไร ข้าก็ไม่ห้ามหรอก เดี๋ยวจะเปิดประตูให้ทันที");
            MoveTo("ama_dun01", 229, 10);
        case 2:
            return;
    }
    //"Ask about the gate"
    Dialog(name, "ประตูนี้นำไปสู่ดินแดนต้องคำสาป เข้าไปแล้วต้องระวังให้ดี ไม่งั้นจะหลงทางและตายอยู่ในนั้น")
    Dialog(name, "ข้าก็ไม่มีอะไรจะบอกมากนัก เพราะไม่เคยเข้าไปเอง แต่พอได้ฟังคำแนะนำจากคนที่เคยเข้าไปมาบ้าง");
    Dialog(name, "ข้อแรก อย่าเชื่อสิ่งที่ตาเห็น ข้าเองก็ไม่ค่อยเข้าใจนัก แต่คิดว่ามันหมายถึงสิ่งที่เห็นอาจไม่ใช่ของจริงทั้งหมด");
    Dialog(name, "ข้อสอง คำสาปมีกฎเกณฑ์ตายตัวของมันเสมอ ผลของมันอาจดูสุ่มไปหมด แต่จริง ๆ แล้วมีเหตุผลอยู่ ที่นี่ก็เหมือนกัน");
    Dialog(name, "ข้าช่วยได้แค่นี้แหละ ยังอยากให้ข้าเปิดประตูให้อยู่ไหม?");
    Option( "เข้าไป", "กลับก่อน")
    switch(Result)
    {
        case 0:
            Dialog(name, "เดี๋ยวข้าเปิดประตูให้ ขอให้โชคดีและปลอดภัยนะ");
            MoveTo("ama_dun01", 229, 10);
        case 1:
            Dialog(name, "ไม่ต้องอายหรอก ข้าเองก็ไม่อยากเข้าไปเหมือนกัน");
            return;
    }
}
