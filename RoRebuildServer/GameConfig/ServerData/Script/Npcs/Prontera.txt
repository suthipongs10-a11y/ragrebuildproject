Npc("prt_in", "Dietrich", "4_M_02", 63, 69, SW) { 
	AddTradedItem(CreateTradedItem("Oridecon").Requires("Rough_Oridecon",5));
	AddTradedItem(CreateTradedItem("Elunium").Requires("Rough_Elunium",5));
OnClick:
	FocusNpc();
	if(InventoryItemCount > MaxItemCount - 1) {
		Dialog(Name, "โอ้โห แบกของมาเยอะเลยนะ! ลดของในกระเป๋าลงหน่อยแล้วค่อยกลับมาใหม่นะ");
		return;
	}
	Dialog(Name, "ผมช่วยสกัด Rough Oridecon กับ Rough Elunium ให้ได้นะ ใช้หินดิบ 5 ก้อน แลกเป็นของบริสุทธิ์ 1 ชิ้น");
	Option("แลกเปลี่ยน", "ยกเลิก");
	if(result == 0) StartItemTrade();
}
