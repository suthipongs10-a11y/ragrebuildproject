//The enchant scribe.
//
//Placed beside the Class Master on purpose: everything a new character does on this map
//happens in that one corner, and a system nobody walks past is a system nobody uses.
//
//Right now this is the shell only - the scrolls, the card dust and the reset scroll do
//not exist yet, so the three menu entries that need them say so rather than opening an
//empty trade window. The dialogue is written for the finished system so the numbers in
//it can be checked against the design before any of it is built.

Npc("prt_fild08", "Enchant Scribe", "2_M_SAGE_OLD", 156, 356, SE, 1, 1)
{
	localstr name = "อาลักษณ์คัมภีร์";
	localstr seenFlag = "EnchantScribeStatus";
OnClick:
	FocusNpc();

	if(GetNpcFlag(seenFlag) == 0)
	{
		Dialog(name, "หยุดก่อนสิเจ้าหนู... ข้าคือ <color=#0066CC>อาลักษณ์คัมภีร์</color> " +
					 "คนที่เปลี่ยนของขยะในกระเป๋าเจ้าให้กลายเป็นพลังบนอุปกรณ์");
		Dialog(name, "ทองคำ เพชรร้าว ตุ๊กตา ของแปลก ๆ ที่หลุดจากกล่องแล้วไม่รู้จะเอาไปทำอะไร " +
					 "เอามาให้ข้าหมด ข้าจะจารเป็น<color=#CC5500>คัมภีร์</color>ให้");
		SetNpcFlag(seenFlag, 1);
	}
	else
	{
		Dialog(name, "กลับมาอีกแล้วหรือ อยากรู้อะไรอีกล่ะ");
	}

	//Two menus rather than one. The old one mixed "tell me about it" with "do it now" in a
	//single list of eight, and the two are not the same kind of thing: one is read once and
	//never again, the other is what somebody walks across a map to press. Doing goes on the
	//front, asking goes behind one door.
	var choice = 0;
	var craft = 0;
	var topic = 0;
	result = 0;
	while(result < 4)
	{
		Option("จารคัมภีร์", "บดการ์ด", "เปิดคู่มือคัมภีร์", "ขอถามอะไรหน่อย", "ไม่มีอะไร");
		choice = result;

		if(choice == 0)
		{
			Dialog(name, "เอาของมาวางตรงนี้ ข้าจะจารให้เดี๋ยวนี้เลย <color=#CC0000>พลาดแล้ววัตถุดิบหายหมดนะ</color> " +
						 "และข้าเลือกไม่ได้ว่าจะออกมาเป็นคัมภีร์ชนิดไหน");
			Option("จารระดับดิน", "จารระดับฟ้า", "จารระดับสวรรค์", "จารระดับตำนาน",
				   "จารคัมภีร์ล้างออพ", "ขอดูสูตรก่อน", "ยังไม่เอา");
			craft = result;

			if(craft == 0) CraftEnchantScroll(1);
			if(craft == 1) CraftEnchantScroll(2);
			if(craft == 2) CraftEnchantScroll(3);
			if(craft == 3) CraftEnchantScroll(4);
			if(craft == 4) CraftBlankScroll();

			//Recipes get their own picker rather than all five at once. Five printed together
			//is forty-odd lines of chat and the one anybody wanted has scrolled off the top.
			if(craft == 5)
			{
				Option("สูตรระดับดิน", "สูตรระดับฟ้า", "สูตรระดับสวรรค์", "สูตรระดับตำนาน",
					   "สูตรคัมภีร์ล้างออพ", "พอแล้ว");

				if(result == 0) ShowEnchantRecipe(1);
				if(result == 1) ShowEnchantRecipe(2);
				if(result == 2) ShowEnchantRecipe(3);
				if(result == 3) ShowEnchantRecipe(4);
				if(result == 4) ShowBlankRecipe();

				if(result < 5)
					Dialog(name, "เขียนไว้ในช่องข้อความแล้ว <color=#55FF55>สีเขียว</color>คือมีพอ " +
								 "<color=#FF5555>สีแดง</color>คือยังขาด");
			}
		}

		if(choice == 1)
		{
			Dialog(name, "การ์ดที่ไม่ได้ใช้มีไหม ข้าบดให้เป็น<color=#CC5500>ผงการ์ด</color>ได้ " +
						 "หนึ่งใบได้หนึ่งผง เอาไว้ทำคัมภีร์ระดับสวรรค์กับตำนาน");
			Dialog(name, "<color=#CC0000>การ์ดบอสข้าไม่บดให้นะ</color> ของแบบนั้นบดทิ้งแล้วเจ้าจะเสียใจ");
			OpenCardGrinder();
			choice = 9;
		}

		//The handbook is a client window, so the scribe only sends the shove that opens it
		//and then bows out - leaving the loop running would put the option menu back on top
		//of the thing the player just asked to read.
		if(choice == 2)
		{
			Dialog(name, "เอ้า ยืมไปอ่านเลย เขียนไว้หมดแล้วว่าระดับไหนใช้ของอะไร หาได้จากมอนตัวไหน แมพอะไร " +
						 "แล้วก็คัมภีร์แต่ละระดับจารออกมาได้ค่าอะไรบ้าง กี่เปอร์เซ็นต์");
			Dialog(name, "กดปุ่ม <color=#CC5500>ระดับถัดไป</color> เพื่อพลิกดูระดับอื่น กดที่ชื่อของเพื่อดูว่ามันดรอปจากตัวไหน");
			OpenEnchantGuide();
			choice = 9;
		}

		if(choice == 3)
		{
			Option("ระบบคัมภีร์คืออะไร", "คัมภีร์มีกี่ระดับ", "คัมภีร์มีกี่ชนิด", "ทำไมจารทับไม่ได้", "พอแล้ว");
			topic = result;

			if(topic == 0)
			{
				Dialog(name, "อุปกรณ์ที่เจ้าใส่อยู่ ใส่<color=#CC5500>ออพชั่นเสริม</color>เพิ่มได้ " +
							 "อาวุธ ชุดเกราะ โล่ รองเท้า ของสวมหัวทั้งสามชั้น และเครื่องประดับ");
				Dialog(name, "วิธีคือเอาวัตถุดิบมาให้ข้าจารเป็นคัมภีร์ก่อน แล้วเจ้าค่อยเอาคัมภีร์ไปใช้กับของที่อยากเสริม " +
							 "ออพที่ได้เป็นการสุ่ม ข้าเองก็บอกล่วงหน้าไม่ได้ว่าจะออกอะไร");
				Dialog(name, "<color=#CC0000>ฟังให้ดี</color> ทั้งตอนจารและตอนใช้ มีโอกาสพลาดทั้งคู่ " +
							 "แต่<b>อุปกรณ์ของเจ้าจะไม่หายและขั้นตีจะไม่ลด</b>ไม่ว่าจะพลาดกี่ครั้ง อย่างมากก็แค่ออพว่างเปล่า");
			}

			if(topic == 1)
			{
				Dialog(name, "มีสี่ระดับ ยิ่งสูงยิ่งจารยาก\n\n" +
							 "<color=#8A6234>ดิน</color> — ออพเดียว ค่าน้อย แต่จารติดเกือบทุกครั้ง\n" +
							 "<color=#1F6E97>ฟ้า</color> — สองออพ เริ่มมีพลังโจมตีกับพลังชีวิตให้สุ่ม");
				Dialog(name, "<color=#8E6E05>สวรรค์</color> — สองออพ ค่าแรงขึ้น ต้องใช้ผงการ์ดด้วย\n" +
							 "<color=#A32E45>ตำนาน</color> — สามออพ ค่าสูงสุด และ<b>จารติดแค่หนึ่งในเจ็ด</b>");
				Dialog(name, "ข้าจารได้ครบทั้งสี่ระดับแล้ว แต่<color=#CC0000>ระดับตำนานต้องเอาคัมภีร์ระดับสวรรค์มาเป็นวัตถุดิบด้วย</color> " +
							 "ชนิดไหนก็ได้ หนึ่งใบ");
				Dialog(name, "คัมภีร์ระดับสวรรค์นอกจากจารเอง ยังหลุดออกมาจากกล่องสมบัติของจอมมารได้อีกทาง " +
							 "แต่<color=#CC0000>กล่องนั้นไม่มีระดับตำนาน</color> ทางเดียวคือมาจารกับข้า");
			}

			if(topic == 2)
			{
				Dialog(name, "คัมภีร์หนึ่งม้วนใช้ได้กับ<color=#CC5500>ตำแหน่งเดียว</color>เท่านั้น ข้าเขียนระบุไว้บนม้วนแล้ว " +
							 "เอาคัมภีร์อาวุธไปใช้กับหมวกไม่ได้ ระดับหนึ่งมีแปดชนิด");
				Dialog(name, "ชนิดที่ออกบ่อย ชนิดละ <color=#CC5500>17 ใน 100</color>\n" +
							 "อาวุธ · ชุดเกราะ · โล่ · รองเท้า · ส่วนหัว Top");
				Dialog(name, "ชนิดที่ออกยาก ชนิดละ <color=#CC0000>5 ใน 100</color>\n" +
							 "ส่วนหัว Middle · ส่วนหัว Low · เครื่องประดับ");
				Dialog(name, "หมวกที่กินหลายชั้นพร้อมกัน ข้านับเป็น<color=#CC5500>ชั้นบนสุดที่มันกิน</color>ชั้นเดียว — " +
							 "หมวกที่กิน Top กับ Middle ต้องใช้คัมภีร์ Top เท่านั้น");
			}

			if(topic == 3)
			{
				Dialog(name, "<color=#CC0000>ของที่มีออพอยู่แล้ว ข้าจารทับให้ไม่ได้</color> " +
							 "กระดาษมันไม่ยอมรับรอยจารสองชั้น");
				Dialog(name, "ถ้าอยากจารใหม่ ต้องล้างของเก่าออกด้วย<color=#CC5500>คัมภีร์ล้างออพ</color>ก่อน " +
							 "ใบนั้นข้าทำให้ได้ อยู่ในเมนูจารคัมภีร์ <b>ทำแล้วสำเร็จเสมอ</b>");
				Dialog(name, "ใช้<color=#CC5500>ถุงเท้าแดง 20 คู่ ผงการ์ด 1 ผง</color> กับ <color=#CC5500>เงินแสน</color> " +
							 "ค่าจ้างแพงหน่อย เพราะมันคือทางเดียวที่จะแก้ของที่จารพลาดไปแล้ว");
			}
		}

		result = choice;
	}

	if(choice != 9)
		Dialog(name, "งั้นก็ตามสบาย ของขยะในกระเป๋าอย่าเพิ่งทิ้งล่ะ");
}
