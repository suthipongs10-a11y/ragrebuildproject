
ServerConfig("ItemAdjustments") {
OnLoadDropData:
	//SWITCHED ON. appsettings.json has RemapDropRates set to true, so every rate in
	//DropData.csv passes through the curve below on the way in and the numbers in that file
	//are the raw material rather than the answer: a card written as 1 is 0.01% there and
	//0.5% in game.
	//
	//This is the rate the project shipped with. It was off for a while, with DropData.csv
	//multiplied by ten in its place, and both halves of that have been undone - the file is
	//back to its original numbers and the curve is back in charge of them. Note that this
	//block also carries the Refine doubling further down; it is part of the same switch and
	//not a setting of its own.

	if(type == Etc && subType == "Quest")
		return rate;

	//Antonio's drops are authored at the rates we actually want to see in game, so they
	//skip the remap below. In DropData.csv 50 means 0.5% and stays 0.5%. Gift Box is in
	//here too, so the two christmas monsters that drop it carry the rate they were
	//already getting after the remap, written out in full instead.
	if(code == "Water_Lily_Hat" || code == "Red_Stocking" || code == "Gift_Box")
		return rate;

	if(type == Etc && subType == "Refine")
	 	rate *= 2; //refine item drop chance is doubled flat out

	//remap drop rates, increasing the rates of low rate drops considerably more than high value items
	if (rate <= 20)			rate = Remap(rate,    1,    20,   50,  200);  //[ 0.01%- 0.2%] -> [ 0.5% - 2.0%] = 50x ~ 8x
	else if (rate <= 80)	rate = Remap(rate,   20,    80,  200,  400);  //[ 0.2% - 0.8%] -> [ 2.0% - 4.0%] = 8x ~ 4x
	else if (rate <= 300)	rate = Remap(rate,   80,   300,  400,  900);  //[ 0.8% - 3.0%] -> [ 4.0% - 9.0%] = 4x ~ 3x
	else if (rate <= 1200)	rate = Remap(rate,  300,  1200,  900,  2400); //[ 3.0% -12.0%] -> [ 9.0% -24.0%] = 3x ~ 2x
	else if (rate <= 10000)	rate = Remap(rate, 1200, 10000, 2400, 10000); //[12.0%-100.0%] -> [24.0%-100.0%] = 2x ~ 1x
	else rate = 10000;

	//What a player drinks keeps the rate the curve gave it. Levelling burns potions and
	//herbs by the hundred, and halving those is felt on every trip out - which is not the
	//scarcity the halving is for. It is aimed at cards, gear and treasure, not at the red
	//potions that get you to them.
	//
	//A usable's subType is its use effect, so these are named by what they do when drunk
	//rather than by a list of item codes that would go stale the first time one was added.
	//Concentration Potion and the badges are deliberately not in the list: they are buffs,
	//not supplies. Treasure boxes have no use effect at all and halve with everything else.
	if(type == Useable && (subType == "RedPotion" || subType == "YellowPotion" || subType == "WhitePotion" || subType == "BluePotion" || subType == "GreenPotion" || subType == "FruitHp" || subType == "FruitSp" || subType == "HealLow" || subType == "HealMid" || subType == "HealHigh"))
		return rate;

	//And then half of that, for everything that is not junk and not a card.
	//
	//The curve above is worth keeping for its shape - it lifts a rare drop far more than a
	//common one, which is what stops the bottom of the table being unreachable - but the
	//height it landed at was too generous for a server that wants gear to mean something.
	//Halving after the remap rather than before keeps the shape and moves only the height:
	//the raw rate on almost every card in DropData.csv is 1, and half of 1 is 0, so halving
	//first would leave nothing for the curve to lift.
	//
	//Cards are the one exception and stay at the height the curve gave them. Every card in
	//DropData.csv is written as 1, which the curve lifts to 50 - a flat 0.5% - and that is the
	//rate the server wants cards to drop at. Halving it again would put them back at 0.25%.
	//
	//Junk is left where it is on purpose. It is what feeds the scribe, the forge and every
	//other recipe, and a recipe that asks for eighty of something is a different kind of
	//grind from a card that asks for one. The three items named above keep their authored
	//rates either way, since they return before they ever reach here.
	if(type != Etc && type != Card)
		rate /= 2;

OnSetItemPurchasePrice:
	//all etc type items (other than those used for refining or fixed NPC sell value) have 2x value
	if(type == Etc && subType == "None")
		price *= 2;
	if(type == Weapon)
		price = price * 3 / 2; //since etc items are worth 2x as much, buying weapons early is really easy, so buy price of weapons is up 50%.

OnSetItemSaleValue:
	//the price variable here is the already modified price set via OnSetItemPurchasePrice.
	if(type != Weapon)
		price /= 2;//sell to npc price is 1/2 the amount you buy it.
	else
		price = price / 3; //even though shop buy prices are 50% higher, they still sell to the npc at the same price as before.
	
	//make ammo sell to npc for 0. Not worth the possible exploit with buying/opening arrow quivers and selling the contents.
	if(type == Ammo)
		return 0; 
}
